-- ===================
-- 3.2.1 Core User Management - UPDATED WITH ALL FIELDS
-- ===================
CREATE TABLE APP_USER (
    user_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    email VARCHAR2(100) UNIQUE NOT NULL,
    password_hash VARCHAR2(255) NOT NULL,
    status VARCHAR2(10) CHECK (status IN ('active','inactive','locked')) NOT NULL,
    created_at DATE NOT NULL,
    -- NEW FIELDS ADDED
    first_name VARCHAR2(50),
    last_name VARCHAR2(50),
    role VARCHAR2(20) CHECK (role IN ('student','faculty','admin','finance','parent')),
    phone VARCHAR2(20),
    avatar_url VARCHAR2(255),
    last_login DATE
);

-- ===================
-- 3.2.2 Academic Structure
-- ===================
CREATE TABLE DEPARTMENT (
    dept_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dept_name VARCHAR2(100) NOT NULL
);

CREATE TABLE PROGRAM (
    program_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR2(100) NOT NULL,
    degree_level VARCHAR2(5) CHECK (degree_level IN ('UG','PG','PhD')) NOT NULL,
    department_id NUMBER,
    CONSTRAINT fk_program_dept FOREIGN KEY (department_id) REFERENCES DEPARTMENT(dept_id)
);

CREATE TABLE COURSE (
    course_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    code VARCHAR2(20) UNIQUE NOT NULL,
    title VARCHAR2(100),
    department_id NUMBER,
    CONSTRAINT fk_course_dept FOREIGN KEY (department_id) REFERENCES DEPARTMENT(dept_id)
);

CREATE TABLE TERM (
    term_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR2(50),
    season VARCHAR2(10) CHECK (season IN ('Fall','Winter','Spring','Summer')),
    start_date DATE,
    end_date DATE
);

-- ===================
-- 3.2.3 Person Management - UPDATED STUDENT TABLE
-- ===================
CREATE TABLE STUDENT (
    student_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id NUMBER UNIQUE,
    program_id NUMBER,
    admission_year NUMBER(4),
    phone VARCHAR2(20),
    -- NEW FIELDS ADDED
    student_code VARCHAR2(20) UNIQUE,
    current_gpa NUMBER(3,2),
    total_credits NUMBER,
    CONSTRAINT fk_student_user FOREIGN KEY (user_id) REFERENCES APP_USER(user_id),
    CONSTRAINT fk_student_program FOREIGN KEY (program_id) REFERENCES PROGRAM(program_id)
);

CREATE TABLE FACULTY (
    faculty_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id NUMBER UNIQUE,
    department_id NUMBER,
    designation VARCHAR2(100),
    CONSTRAINT fk_faculty_user FOREIGN KEY (user_id) REFERENCES APP_USER(user_id),
    CONSTRAINT fk_faculty_dept FOREIGN KEY (department_id) REFERENCES DEPARTMENT(dept_id)
);

CREATE TABLE PARENT (
    parent_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR2(100) NOT NULL,
    relation_preference VARCHAR2(50)
);

CREATE TABLE STUDENT_PARENT (
    student_id NUMBER,
    parent_id NUMBER,
    relationship VARCHAR2(10) CHECK (relationship IN ('father','mother','guardian','other')),
    linked_at DATE,
    PRIMARY KEY (student_id, parent_id),
    CONSTRAINT fk_sp_student FOREIGN KEY (student_id) REFERENCES STUDENT(student_id),
    CONSTRAINT fk_sp_parent FOREIGN KEY (parent_id) REFERENCES PARENT(parent_id)
);

-- ===================
-- 3.2.4 Academic Operations
-- ===================
CREATE TABLE COURSE_OFFERING (
    offering_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    course_id NUMBER,
    faculty_id NUMBER,
    term_id NUMBER,
    section VARCHAR2(10),
    capacity NUMBER,
    schedule VARCHAR2(50),
    room VARCHAR2(50),
    CONSTRAINT fk_offering_course FOREIGN KEY (course_id) REFERENCES COURSE(course_id),
    CONSTRAINT fk_offering_faculty FOREIGN KEY (faculty_id) REFERENCES FACULTY(faculty_id),
    CONSTRAINT fk_offering_term FOREIGN KEY (term_id) REFERENCES TERM(term_id)
);

CREATE TABLE ENROLLMENT (
    enrollment_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    student_id NUMBER,
    offering_id NUMBER,
    term_id NUMBER,
    enrolled_at DATE,
    status VARCHAR2(10) CHECK (status IN ('active','dropped','completed','failed')),
    CONSTRAINT fk_enroll_student FOREIGN KEY (student_id) REFERENCES STUDENT(student_id),
    CONSTRAINT fk_enroll_offering FOREIGN KEY (offering_id) REFERENCES COURSE_OFFERING(offering_id),
    CONSTRAINT fk_enroll_term FOREIGN KEY (term_id) REFERENCES TERM(term_id)
);

CREATE TABLE ATTENDANCE (
    attendance_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    enrollment_id NUMBER,
    class_date DATE,
    status VARCHAR2(10) CHECK (status IN ('present','absent','excused','manual')),
    CONSTRAINT fk_attendance_enroll FOREIGN KEY (enrollment_id) REFERENCES ENROLLMENT(enrollment_id)
);

CREATE TABLE ASSESSMENT (
    assessment_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    offering_id NUMBER,
    title VARCHAR2(100),
    max_score NUMBER,
    weight NUMBER(5,2),
    due_date DATE,
    CONSTRAINT fk_assessment_offering FOREIGN KEY (offering_id) REFERENCES COURSE_OFFERING(offering_id)
);

CREATE TABLE GRADE (
    grade_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    enrollment_id NUMBER,
    assessment_id NUMBER,
    grade NUMBER(5,2),
    grade_date DATE,
    grader_id NUMBER,
    CONSTRAINT fk_grade_enrollment FOREIGN KEY (enrollment_id) REFERENCES ENROLLMENT(enrollment_id),
    CONSTRAINT fk_grade_assessment FOREIGN KEY (assessment_id) REFERENCES ASSESSMENT(assessment_id),
    CONSTRAINT fk_grade_grader FOREIGN KEY (grader_id) REFERENCES FACULTY(faculty_id)
);

-- ===================
-- 3.2.5 Financial Management
-- ===================
CREATE TABLE INVOICE (
    invoice_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    student_id NUMBER,
    term_id NUMBER,
    amount NUMBER(10,2),
    status VARCHAR2(10) CHECK (status IN ('unpaid','partial','paid','voided')),
    due_date DATE,
    created_at DATE,
    CONSTRAINT fk_invoice_student FOREIGN KEY (student_id) REFERENCES STUDENT(student_id),
    CONSTRAINT fk_invoice_term FOREIGN KEY (term_id) REFERENCES TERM(term_id)
);

CREATE TABLE PAYMENT (
    payment_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    invoice_id NUMBER,
    amount NUMBER(10,2),
    paid_on DATE,
    method VARCHAR2(20) CHECK (method IN ('card','netbanking','cheque','cash','upi')),
    notes VARCHAR2(255),
    CONSTRAINT fk_payment_invoice FOREIGN KEY (invoice_id) REFERENCES INVOICE(invoice_id)
);

-- ===================
-- 3.2.6 Communication System
-- ===================
CREATE TABLE NOTIFICATION (
    notification_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id NUMBER,
    type VARCHAR2(10) CHECK (type IN ('email','sms','push','system')),
    content VARCHAR2(255),
    created_at DATE,
    read_at DATE,
    CONSTRAINT fk_notification_user FOREIGN KEY (user_id) REFERENCES APP_USER(user_id)
);

COMMIT;

PROMPT PARENT-STUDENT RELATIONSHIPS:
PROMPT ===============================================
SELECT p.parent_id, p.name as parent_name, COUNT(sp.student_id) as children_count
FROM parent p
LEFT JOIN student_parent sp ON p.parent_id = sp.parent_id
GROUP BY p.parent_id, p.name
ORDER BY p.parent_id;
